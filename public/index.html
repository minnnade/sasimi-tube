// 2. 選択した動画のgooglevideo URLを取得して再生する関数 async function playVideo(videoId) { const videoElement = document.getElementById('player'); try { // サーバーからgooglevideoの生URLをもらう const response = await fetch(`/api/stream-url?id=${videoId}`); const data = await response.json(); if (!data.url) { alert('ストリームURLの取得に失敗しました'); return; } // 💡 取得した https://*://... をそのままvideoタグにセット！ videoElement.src = data.url; window.scrollTo({ top: 0, behavior: 'smooth' }); } catch (error) { console.error('Error fetching stream url:', error); alert('エラーが発生しました'); } }
